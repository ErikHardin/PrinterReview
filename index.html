function loadExcel(inp){

  const file=inp.files[0];if(!file)return;

  const errEl=document.getElementById('setupErr');errEl.classList.remove('show');

  const reader=new FileReader();

  reader.onload=e=>{

    try{

      const wb=XLSX.read(e.target.result,{type:'array'});

      const rows=XLSX.utils.sheet_to_json(wb.Sheets[wb.SheetNames[0]],{header:1,defval:''});

      if(rows.length<2)throw new Error('Spreadsheet appears to be empty.');

      const cm={};

      rows[0].forEach((h,i)=>{const k=nk(h);for(const[f,al]of Object.entries(CM)){if(!(f in cm)&&al.some(a=>nk(a)===k))cm[f]=i;}});

      const miss=['model','dnumber','bnumber','building','floor','room'].filter(f=>!(f in cm));

      if(miss.length)throw new Error('Could not find columns: '+miss.join(', '));

      const g=(row,f)=>cm[f]!==undefined?String(row[cm[f]]||'').trim():'';

      const loaded=[];

      rows.slice(1).forEach((row,i)=>{const m=g(row,'model');if(!m)return;loaded.push({id:i+1,institute:g(row,'institute'),dept:g(row,'dept'),building:g(row,'building'),floor:g(row,'floor'),room:g(row,'room'),model:m,dnumber:g(row,'dnumber'),bnumber:g(row,'bnumber'),avgPages:parseFloat(g(row,'avgPages'))||0});});

      if(!loaded.length)throw new Error('No printer rows found.');

      printers=loaded;decisions={};save();

      const url=encodeToUrl(printers);

      document.getElementById('shareUrl').textContent=url;

      document.getElementById('btnShare').style.display='';

      openDlg('shareDlg');

      askName();

    }catch(err){errEl.textContent='⚠️ '+err.message;errEl.classList.add('show');}

  };

  reader.readAsArrayBuffer(file);

}
